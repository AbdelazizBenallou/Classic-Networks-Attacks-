Step 02 :
config the dhcp server to deffuse the ps for each vlan =

give the dhcp server a static ip in his vlan  + the gateway : 

that mean the router have ip static o his interface 

ok enter to /etc/dhcp//dhcpd.conf

subnet 192.168.1.0 netmask 255.255.255.0 {
  range 192.168.1.100 192.168.1.200;
  option routers 192.168.1.254;
}

subnet 192.168.10.0 netmask 255.255.255.0 {
  range 192.168.10.100 192.168.1.200;
  option routers 192.168.10.254;
}

subnet 192.168.20.0 netmask 255.255.255.0 {
  range 192.168.20.100 192.168.20.200;
  option routers 192.168.20.254;
}


subnet 192.168.99.0 netmask 255.255.255.255 {
  range 192.168.99.100 192.168.99.200;
  option routers 192.168.99.254;
}


subnet 192.168.100.0 netmask 255.255.255.0 {
  range 192.168.100.100 192.168.100.200;
  option routers 192.168.100.254;
}



after this just run the dhcp server
